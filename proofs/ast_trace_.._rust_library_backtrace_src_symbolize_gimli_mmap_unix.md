impl Mmap {

        let ptr = unsafe {
            mmap64(
                ptr::null_mut(),
                len,
                libc::PROT_READ,
                libc::MAP_PRIVATE,
                file.as_raw_fd(),
                offset.try_into().ok()?,
            )
        };

            return None;
        }
